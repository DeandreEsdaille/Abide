cmake_minimum_required(VERSION 3.22)
project(Abide VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(APPLE)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS")
endif()
include(FetchContent)
if(DEFINED JUCE_PATH)
 add_subdirectory("${JUCE_PATH}" JUCE)
else()
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW FALSE)
 FetchContent_MakeAvailable(JUCE)
endif()
juce_add_binary_data(AbideArt SOURCES Assets/atmosphere.jpg)
juce_add_plugin(Abide COMPANY_NAME "Gimme That Juice" BUNDLE_ID "com.gimmethatjuice.abide" PLUGIN_MANUFACTURER_CODE GtJu PLUGIN_CODE Abid PRODUCT_NAME "Abide" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE FORMATS VST3 Standalone VST3_CATEGORIES Instrument Synth)
target_sources(Abide PRIVATE Source/Plugin.cpp)
target_compile_definitions(Abide PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Abide PRIVATE AbideArt juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
add_executable(engine_tests tests/engine_tests.cpp)
add_test(NAME engine_tests COMMAND engine_tests)
juce_add_console_app(AbideSmoke PRODUCT_NAME "Abide Smoke Test")
target_sources(AbideSmoke PRIVATE tests/processor_smoke.cpp)
target_link_libraries(AbideSmoke PRIVATE Abide juce::juce_audio_processors)
add_test(NAME processor_smoke COMMAND AbideSmoke)
